\documentclass[tikz,border=5pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning}

\begin{document}
\begin{tikzpicture}[
    >=stealth,              % for arrow tips
    node distance=2.8cm,    % adjusts spacing between nodes
    on grid,
    auto
]

% 1) Circle for the prior distribution
\node[circle, draw, minimum width=2.2cm, align=center] (prior)
{Prior\\$p_{\theta}(z)$};

% 2) Decoder to the right of the prior
\node[draw, rectangle, align=center, right=of prior] (decoder)
{Decoder\\$p_{\theta}(x\mid z)$};

% 3) "x-space" node (squiggly) to the right of the decoder
\node[draw, rectangle, align=center, minimum width=2.4cm,
      minimum height=2.4cm, right=of decoder] (xspace) {};

% Draw a simple squiggle inside the xspace node
\begin{scope}[on background layer]
\draw[thick]
  ($(xspace.south west) + (0.4,0.6)$)
  to[out=50, in=130]
  ($(xspace.south west) + (1.0,1.2)$)
  to[out=-50, in=180]
  ($(xspace.south west) + (1.4,0.6)$);
\end{scope}

% Label for the x-space node
\node at (xspace) {$x$-space};

% 4) Dataset node to the right of xspace
\node[draw, rectangle, align=center, right=of xspace] (dataset)
{Dataset\\$\mathcal{D}$};

% 5) Encoder node above xspace
\node[draw, rectangle, align=center, above=of xspace] (encoder)
{Encoder\\$q_{\phi}(z\mid x)$};

% Arrows
\draw[->] (prior)   -- (decoder);
\draw[->] (decoder) -- (xspace);
\draw[->] (xspace)  -- (dataset);
\draw[->] (xspace)  -- (encoder);
\draw[->] (encoder) -- (prior);

\end{tikzpicture}
\end{document}
